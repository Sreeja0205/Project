import sys
import os
from pyspark.sql import SparkSession

PROJECT_ROOT = "/mnt/c/Training/capstone/Customer_Engagement"
if PROJECT_ROOT not in sys.path:
    sys.path.append(PROJECT_ROOT)

from schemas.products_schema import EXPECTED_SCHEMA
from src.bronze_engine import ingest_to_bronze

spark = SparkSession.builder \
    .appName("Bronze_Ingestion_Products") \
    .config("spark.sql.session.timeZone", "UTC") \
    .config("spark.sql.catalog.datalake", "org.apache.iceberg.spark.SparkCatalog") \
    .config("spark.sql.catalog.datalake.type", "hadoop") \
    .config("spark.sql.catalog.datalake.warehouse", "hdfs:///localhost:9000/user/sreeja/capstone_bronze/warehouse") \
    .getOrCreate()
spark.sparkContext.setLogLevel("WARN")

source = "file:///mnt/c/Training/capstone/capstone_data/api-*/csv/products/products/*.csv"

valid_df, bad_df = ingest_to_bronze(
    spark=spark,
    dataset_name="products",
    source_path=source,
    expected_schema=EXPECTED_SCHEMA,
    source_format="csv"
)

print(f"Valid records processed: {valid_df.count()}")
print(f"Quarantined records processed: {bad_df.count()}")
