from pyspark.sql import types as T

relationship_schema = T.StructType([
    T.StructField("id", T.LongType(), False),
    T.StructField("name", T.StringType(), False),
])

EXPECTED_SCHEMA = T.StructType([
    T.StructField("id", T.LongType(), False),
    T.StructField("move_id", relationship_schema, True),
    T.StructField("picking_id", relationship_schema, True),
    T.StructField("product_id", relationship_schema, True),
    T.StructField("quantity", T.DecimalType(38, 18), True),
    T.StructField("location_id", relationship_schema, True),
    T.StructField("location_dest_id", relationship_schema, True),
    T.StructField("write_date", T.TimestampType(), True),
])
