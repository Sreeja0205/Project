])

    T.StructField("categ_id", relationship_schema, True),
    T.StructField("uom_id", relationship_schema, True),
    T.StructField("list_price", T.DecimalType(38, 18), True),
    T.StructField("create_date", T.TimestampType(), True),
    T.StructField("write_date", T.TimestampType(), True),
    T.StructField("active", T.BooleanType(), True),
    T.StructField("qty_available", T.DecimalType(38, 18), True),
    T.StructField("virtual_available", T.DecimalType(38, 18), True),
    T.StructField("product_tmpl_id", relationship_schema, True),
